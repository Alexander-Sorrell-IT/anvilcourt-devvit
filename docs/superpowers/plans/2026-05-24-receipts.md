# Receipts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Devvit Web background app that, on every Reddit content removal (mod, AutoMod silent filter, AutoMod remove, spam), auto-sends the author a rule-cited explanation with an appeal path and writes a searchable mod-only audit log — deterministically, with no LLM and no auto-removal.

**Architecture:** One pipeline — triggers → dedup → reason resolver → explanation composer → delivery (in-place comment + modmail) → audit. The **core logic is pure TypeScript** (`src/core/`, no Devvit imports → unit-testable with vitest). The **Devvit glue is thin adapters** (`src/server/`) wiring `devvit.json` `/internal/...` endpoints to the core, verified by playtest. The bot routes appeals to humans; it never adjudicates.

**Tech Stack:** TypeScript, Node 22, Devvit Web (`devvit.json` + Hono server, `@devvit/web/server` for `reddit`/`redis`/`settings`/`scheduler`), Redis (per-sub), vitest for unit tests.

**Spec:** `docs/superpowers/specs/2026-05-24-receipts-design.md`

If any differ, adjust the thin adapter — the pure core is unaffected.

```ts
import type { ResolvedReason, ItemType } from './types';

export interface ComposeInput {
  reason: ResolvedReason;
  itemType: ItemType;
  subreddit: string;
  template: string;            // tokens: {{rule}} {{reason}} {{itemType}} {{subreddit}} {{appeal}}
  appealInstructions: string;
  appealsEnabled: boolean;
}

export function composeExplanation(i: ComposeInput): string {
  const appeal = i.appealsEnabled ? i.appealInstructions : '';
  return i.template
    .replaceAll('{{rule}}', i.reason.ruleRef ?? '')
    .replaceAll('{{reason}}', i.reason.text)
    .replaceAll('{{itemType}}', i.itemType)
    .replaceAll('{{subreddit}}', i.subreddit)
    .replaceAll('{{appeal}}', appeal)
    .trim();
}
```

```ts
import type { RemovalEvent, ResolvedReason } from './types';

export interface ReasonLookups {
  modLogReason: (itemId: string) => Promise<string | undefined>;
  ruleConfigText: (ruleRef: string | undefined) => Promise<string | undefined>;
  deriveReason: (event: RemovalEvent) => Promise<string | undefined>;
}

const GENERIC = 'Your content was removed by the moderators of this community.';
const ok = (s?: string) => !!s && s.trim().length > 0;

export async function resolveReason(event: RemovalEvent, l: ReasonLookups): Promise<ResolvedReason> {
  if (ok(event.rawReason)) return { text: event.rawReason!.trim(), tier: 'filter-reason', ruleRef: event.ruleRef };
  const ml = await l.modLogReason(event.itemId);
  if (ok(ml)) return { text: ml!.trim(), tier: 'mod-log', ruleRef: event.ruleRef };
  const rc = await l.ruleConfigText(event.ruleRef);
  if (ok(rc)) return { text: rc!.trim(), tier: 'rule-config', ruleRef: event.ruleRef };
  const dv = await l.deriveReason(event);
  if (ok(dv)) return { text: dv!.trim(), tier: 'derived' };
  return { text: GENERIC, tier: 'generic' };
}
```

```ts
import type { ReceiptRecord } from './types';

export function isAppealReply(o: { hasRecentReceipt: boolean; isFromModerator: boolean; isInternal: boolean }): boolean {
  if (o.isFromModerator || o.isInternal) return false;
  return o.hasRecentReceipt;
}

export function pickReceiptToFlag(records: ReceiptRecord[], now: number, windowMs: number): ReceiptRecord | undefined {
  return records
    .filter(r => r.appealStatus === 'none' && now - r.ts <= windowMs)
    .sort((a, b) => b.ts - a.ts)[0];
}
```

```ts
import type { RemovalSource } from './types';

export const DEFAULT_TEMPLATE =
  'Your {{itemType}} in r/{{subreddit}} was removed.\n\nReason: {{reason}}\n\n{{appeal}}';
export const DEFAULT_APPEAL = 'If you think this was a mistake, reply to this message and a moderator will review it. (This is an automated notice; a human makes the final call.)';

export interface ReceiptsConfig {
  deliveryChannel: 'comment' | 'modmail' | 'both' | 'off';
  appealsEnabled: boolean;
  template: string;
  appealInstructions: string;
  perReasonOptOut: string[];
  explainSources: { modRemovals: boolean; automodFilter: boolean; automodRemove: boolean; spamFilter: boolean };
}

const bool = (v: unknown, d: boolean) => (typeof v === 'boolean' ? v : d);
const str = (v: unknown, d: string) => (typeof v === 'string' && v.length ? v : d);

export function parseConfig(raw: Record<string, unknown>): ReceiptsConfig {
  return {
    deliveryChannel: (str(raw.deliveryChannel, 'both') as ReceiptsConfig['deliveryChannel']),
    appealsEnabled: bool(raw.appealsEnabled, true),
    template: str(raw.messageTemplate, DEFAULT_TEMPLATE),
    appealInstructions: str(raw.appealInstructions, DEFAULT_APPEAL),
    perReasonOptOut: str(raw.perReasonOptOut, '').split(',').map(s => s.trim()).filter(Boolean),
    explainSources: {
      modRemovals: bool(raw.explainModRemovals, true),
      automodFilter: bool(raw.explainAutomodFilter, true),
      automodRemove: bool(raw.explainAutomodRemove, true),
      spamFilter: bool(raw.explainSpamFilter, false),
    },
  };
}

export function shouldExplain(source: RemovalSource, c: ReceiptsConfig): boolean {
  if (c.deliveryChannel === 'off') return false;
  return source === 'mod' ? c.explainSources.modRemovals
    : source === 'automod-filter' ? c.explainSources.automodFilter
    : source === 'automod-remove' ? c.explainSources.automodRemove
    : c.explainSources.spamFilter;
}
```

```ts
import { redis } from '@devvit/web/server';
import type { ReceiptRecord, AppealStatus } from '../core/types';

const SEEN = (id: string) => `receipt:seen:${id}`;
const ITEM = (id: string) => `receipt:item:${id}`;
const USER = (u: string) => `receipt:user:${u.toLowerCase()}`;
const RECENT = 'receipt:recent';
const RECENT_MAX = 500;
const SEEN_TTL_SECONDS = 3600;

/** Returns true if THIS call is the first to see itemId (claims it); false if already claimed. */
export async function claimUnseen(itemId: string): Promise<boolean> {
  // Confirm option names in Task 2; expected: set with NX + expiration.
  const res = await redis.set(SEEN(itemId), '1', { nx: true, expiration: SEEN_TTL_SECONDS });
  return res !== null; // non-null => we set it (was unseen)
}

export async function writeRecord(r: ReceiptRecord): Promise<void> {
  await redis.hSet(ITEM(r.itemId), {
    author: r.author, itemType: r.itemType, source: r.source,
    reasonText: r.reasonText, reasonTier: r.reasonTier, ruleRef: r.ruleRef ?? '',
    deliveredVia: r.deliveredVia, modName: r.modName ?? '', ts: String(r.ts),
    appealStatus: r.appealStatus, appealedAt: r.appealedAt ? String(r.appealedAt) : '',
  });
  await redis.zAdd(USER(r.author), { member: r.itemId, score: r.ts });
  await redis.zAdd(RECENT, { member: r.itemId, score: r.ts });
  // trim recent to newest RECENT_MAX
  const count = await redis.zCard(RECENT);
  if (count > RECENT_MAX) await redis.zRemRangeByRank(RECENT, 0, count - RECENT_MAX - 1);
}

export async function getRecord(itemId: string): Promise<ReceiptRecord | undefined> {
  const h = await redis.hGetAll(ITEM(itemId));
  if (!h || !h.author) return undefined;
  return {
    itemId, author: h.author, itemType: h.itemType as ReceiptRecord['itemType'],
    source: h.source as ReceiptRecord['source'], reasonText: h.reasonText,
    reasonTier: h.reasonTier as ReceiptRecord['reasonTier'], ruleRef: h.ruleRef || undefined,
    deliveredVia: h.deliveredVia, modName: h.modName || undefined, ts: Number(h.ts),
    appealStatus: (h.appealStatus as AppealStatus) || 'none',
    appealedAt: h.appealedAt ? Number(h.appealedAt) : undefined,
  };
}

export async function getUserRecords(username: string, limit = 25): Promise<ReceiptRecord[]> {
  const ids = await redis.zRange(USER(username), 0, limit - 1, { reverse: true, by: 'rank' });
  const out: ReceiptRecord[] = [];
  for (const e of ids) { const r = await getRecord(typeof e === 'string' ? e : e.member); if (r) out.push(r); }
  return out;
}

export async function getRecent(limit = 25): Promise<ReceiptRecord[]> {
  const ids = await redis.zRange(RECENT, 0, limit - 1, { reverse: true, by: 'rank' });
  const out: ReceiptRecord[] = [];
  for (const e of ids) { const r = await getRecord(typeof e === 'string' ? e : e.member); if (r) out.push(r); }
  return out;
```

```ts
import { reddit } from '@devvit/web/server';

export interface DeliveryResult { via: string }

/** Post a stickied+distinguished mod comment on the (removed) item. Returns true on success. */
export async function deliverComment(itemId: string, body: string): Promise<boolean> {
  try {
    const comment = await reddit.submitComment({ id: itemId, text: body });
    await comment.distinguish(true); // sticky + distinguish
    return true;
  } catch (e) {
    console.error(`[receipts] comment delivery failed for ${itemId}:`, e);
    return false;
  }
}

/** DM the user as the subreddit via modmail (creates the two-way appeal thread). */
export async function deliverModmail(subreddit: string, username: string, subject: string, body: string): Promise<boolean> {
  try {
    await reddit.modMail.createConversation({ subredditName: subreddit, to: username, subject, body, isAuthorHidden: true });
    return true;
  } catch (e) {
    console.error(`[receipts] modmail delivery failed for u/${username}:`, e);
    return false;
  }
}

export async function deliver(opts: {
  channel: 'comment' | 'modmail' | 'both' | 'off';
  itemId: string; itemType: 'post' | 'comment'; subreddit: string; username: string; body: string;
}): Promise<DeliveryResult> {
  const parts: string[] = [];
  const wantComment = (opts.channel === 'comment' || opts.channel === 'both');
  const wantModmail = (opts.channel === 'modmail' || opts.channel === 'both');
  // Comment-in-place is reliable for posts; for comments it may fail — modmail is the fallback.
  if (wantComment && await deliverComment(opts.itemId, opts.body)) parts.push('comment');
  if (wantModmail && await deliverModmail(opts.subreddit, opts.username, 'Your content was removed', opts.body)) parts.push('modmail');
  return { via: parts.length ? parts.join('+') : 'none' };
}
```

### Task 11: processRemoval orchestrator

**Files:**
- Create: `src/server/pipeline.ts`

```ts
import { settings } from '@devvit/web/server';
import type { RemovalEvent, ReceiptRecord } from '../core/types';
import { parseConfig, shouldExplain } from '../core/config';
import { resolveReason } from '../core/reasonResolver';
import { composeExplanation } from '../core/explanationComposer';
import { makeReasonLookups } from './reasonSources';
import { deliver } from './delivery';
import { claimUnseen, writeRecord } from './audit';

export async function processRemoval(event: RemovalEvent): Promise<void> {
  // 1. Dedup (both ModAction and filter triggers can fire for one item)
  if (!(await claimUnseen(event.itemId))) return;

  // 2. Config + source gating
  const cfg = parseConfig(await settings.getAll());
  if (!shouldExplain(event.source, cfg)) return;

  // 3. Resolve reason (4-tier)
  const reason = await resolveReason(event, makeReasonLookups(event.subreddit));

  // 4. Per-reason opt-out (don't tip off e.g. spam)
  const optedOut = cfg.perReasonOptOut.some(o => reason.ruleRef === o || reason.text.toLowerCase().includes(o.toLowerCase()));

  // 5. Compose + deliver
  let deliveredVia = 'suppressed';
  if (!optedOut) {
    const body = composeExplanation({
      reason, itemType: event.itemType, subreddit: event.subreddit,
      template: cfg.template, appealInstructions: cfg.appealInstructions, appealsEnabled: cfg.appealsEnabled,
    });
    deliveredVia = (await deliver({
      channel: cfg.deliveryChannel, itemId: event.itemId, itemType: event.itemType,
      subreddit: event.subreddit, username: event.author, body,
    })).via;
  }

  // 6. Audit (always — even if suppressed/failed)
  const record: ReceiptRecord = {
    itemId: event.itemId, author: event.author, itemType: event.itemType, source: event.source,
    reasonText: reason.text, reasonTier: reason.tier, ruleRef: reason.ruleRef,
    deliveredVia, modName: event.modName, ts: event.ts, appealStatus: 'none',
  };
  await writeRecord(record);
}
```
